# ==============================================================================
# O3 Car Spaa — Production Crontab Schedule (Hostinger VPS)
# Install via: crontab deploy/crontab.txt
# ==============================================================================

# 1. Database + uploads backup twice a day (02:00 and 14:00 IST), with off-server copy when BACKUP_REMOTE is set
0 2,14 * * * /bin/bash /var/www/o3-car-spaa/scripts/backup.sh >> /var/log/o3-backup.log 2>&1

# 2. Weekly Automated Backup Restore & Row Integrity Verification (Every Sunday at 03:00 AM IST)
0 3 * * 0 /bin/bash /var/www/o3-car-spaa/scripts/test-restore.sh >> /var/log/o3-restore-test.log 2>&1

# 3. Let's Encrypt SSL Certificate Auto-Renewal (Twice daily at 00:00 and 12:00)
0 0,12 * * * certbot renew --quiet --post-hook "systemctl reload nginx"

# 4. PM2 Log Rotation & Cleanup (Daily at midnight)
0 0 * * * /usr/bin/pm2 flush > /dev/null 2>&1
